Case Study 03: Enterprise Sales Data Warehouse — 5 BigQuery SQL Analytical Queries

1. Monthly Sales & YTD Running Totals (SUM() OVER Window Function) 
WITH Monthly_Sales AS (
  SELECT
    EXTRACT(YEAR FROM OrderDate) AS Sales_Year,
    EXTRACT(MONTH FROM OrderDate) AS Sales_Month,
    ROUND(SUM(SalesAmount), 2) AS Monthly_Revenue
  FROM `enterprise_dwh.Fact_Internet_Sales`
  GROUP BY Sales_Year, Sales_Month
)
SELECT
  Sales_Year,
  Sales_Month,
  Monthly_Revenue,
  ROUND(SUM(Monthly_Revenue) OVER (
    PARTITION BY Sales_Year
    ORDER BY Sales_Month
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ), 2) AS YTD_Running_Total
FROM Monthly_Sales
ORDER BY Sales_Year, Sales_Month;
2. Multi-Channel Fact Consolidation (UNION ALL + 3 Product Dimension Joins) 
WITH All_Channel_Sales AS (
  SELECT ProductKey, OrderDate, 'Internet' AS Channel, OrderQuantity, SalesAmount
  FROM `enterprise_dwh.Fact_Internet_Sales`
  UNION ALL
  SELECT ProductKey, OrderDate, 'Reseller' AS Channel, OrderQuantity, SalesAmount
  FROM `enterprise_dwh.Fact_Reseller_Sales`
)
SELECT
  c.EnglishProductCategoryName AS Category,
  sc.EnglishProductSubcategoryName AS Subcategory,
  p.EnglishProductName AS Product_Name,
  s.Channel,
  SUM(s.OrderQuantity) AS Total_Units,
  ROUND(SUM(s.SalesAmount), 2) AS Total_Revenue
FROM All_Channel_Sales s
INNER JOIN `enterprise_dwh.Dim_Product` p ON s.ProductKey = p.ProductKey
LEFT JOIN `enterprise_dwh.Dim_Product_Subcategory` sc ON p.ProductSubcategoryKey = sc.ProductSubcategoryKey
LEFT JOIN `enterprise_dwh.Dim_Product_Category` c ON sc.ProductCategoryKey = c.ProductCategoryKey
GROUP BY Category, Subcategory, Product_Name, s.Channel
ORDER BY Total_Revenue DESC;
3. Regional Product Revenue Share & High/Low Flag (Subquery + CASE WHEN) 
SELECT
  r.SalesTerritoryRegion AS Region,
  p.EnglishProductName AS Product_Name,
  ROUND(SUM(f.SalesAmount), 2) AS Regional_Product_Sales,
  ROUND(SUM(f.SalesAmount) * 100.0 / (
    SELECT SUM(s2.SalesAmount)
    FROM `enterprise_dwh.Fact_Internet_Sales` s2
    WHERE s2.SalesTerritoryKey = f.SalesTerritoryKey
  ), 2) AS Region_Revenue_Share_Pct,
  CASE
    WHEN SUM(f.SalesAmount) >= AVG(SUM(f.SalesAmount)) OVER (PARTITION BY f.SalesTerritoryKey)
      THEN 'High Contributor'
    ELSE 'Low Contributor'
  END AS Performance_Flag
FROM `enterprise_dwh.Fact_Internet_Sales` f
INNER JOIN `enterprise_dwh.Dim_Product` p ON f.ProductKey = p.ProductKey
INNER JOIN `enterprise_dwh.Dim_Sales_Territory` r ON f.SalesTerritoryKey = r.SalesTerritoryKey
GROUP BY f.SalesTerritoryKey, Region, Product_Name
ORDER BY Region, Regional_Product_Sales DESC;
4. Working-Day Classification Excluding Sundays & Public Holidays 
SELECT
  cal.FullDateAlternateKey AS Calendar_Date,
  cal.EnglishDayNameOfWeek AS Day_Name,
  hol.Holiday_Name,
  CASE
    WHEN EXTRACT(DAYOFWEEK FROM cal.FullDateAlternateKey) = 1 THEN 'Non-Working (Sunday)'
    WHEN hol.Holiday_Date IS NOT NULL THEN 'Non-Working (Public Holiday)'
    ELSE 'Working Day (SLA Active)'
  END AS Operational_Day_Type
FROM `enterprise_dwh.Dim_Calendar` cal
LEFT JOIN `enterprise_dwh.Dim_Public_Holidays` hol
  ON cal.FullDateAlternateKey = hol.Holiday_Date
ORDER BY Calendar_Date;
5. Top-3 Department Salary Ranking (DENSE_RANK() Window Function) 
WITH Ranked_Salaries AS (
  SELECT
    d.DepartmentName AS Department,
    CONCAT(e.FirstName, ' ', e.LastName) AS Employee_Name,
    e.Title AS Job_Title,
    e.BaseRate AS Salary_Rate,
    DENSE_RANK() OVER (
      PARTITION BY d.DepartmentName
      ORDER BY e.BaseRate DESC
    ) AS Salary_Rank
  FROM `enterprise_dwh.Dim_Employee` e
  INNER JOIN `enterprise_dwh.Dim_Department` d ON e.DepartmentKey = d.DepartmentKey
  WHERE e.CurrentFlag = 1
)
SELECT Department, Employee_Name, Job_Title, Salary_Rate, Salary_Rank
FROM Ranked_Salaries
WHERE Salary_Rank <= 3
ORDER BY Department, Salary_Rank;
